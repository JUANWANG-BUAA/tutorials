cmake_minimum_required(VERSION 3.24)
project(rust_ffi_bridge VERSION 1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_program(CARGO cargo REQUIRED)
find_package(Threads REQUIRED)
option(RB_TEST_HOOKS "Compile intentional panic probe (test builds only)" ON)
set(RUST_TARGET_DIR "${CMAKE_BINARY_DIR}/cargo-target")
set(RUST_FEATURES "")
if(RB_TEST_HOOKS)
    set(RUST_FEATURES --features test-hooks)
endif()
if(WIN32)
    message(FATAL_ERROR "This native integration is validated on macOS; Windows system libraries are not specified")
endif()
set(RUST_ARCHIVE "${RUST_TARGET_DIR}/release/libffi_bridge.a")
add_custom_command(OUTPUT "${RUST_ARCHIVE}"
    COMMAND "${CARGO}" build --offline --locked --release
        --manifest-path "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml"
        --target-dir "${RUST_TARGET_DIR}" ${RUST_FEATURES}
    DEPENDS src/lib.rs Cargo.toml Cargo.lock
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    VERBATIM)
add_custom_target(rust_component ALL DEPENDS "${RUST_ARCHIVE}")
add_library(rust_bridge STATIC IMPORTED GLOBAL)
set_target_properties(rust_bridge PROPERTIES IMPORTED_LOCATION "${RUST_ARCHIVE}")
add_dependencies(rust_bridge rust_component)
foreach(program demo tests benchmark)
    add_executable(bridge_${program} "c/${program}.c")
    target_include_directories(bridge_${program} PRIVATE include)
    # Tests and checksums must remain active even under Release's -DNDEBUG.
    target_compile_options(bridge_${program} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(bridge_${program} PRIVATE rust_bridge Threads::Threads ${CMAKE_DL_LIBS})
    if(APPLE)
        target_link_libraries(bridge_${program} PRIVATE "-framework Security" "-framework CoreFoundation" "-liconv")
    elseif(UNIX)
        target_link_libraries(bridge_${program} PRIVATE m)
    endif()
endforeach()
if(RB_TEST_HOOKS)
    target_compile_definitions(bridge_tests PRIVATE RB_TEST_HOOKS=1)
endif()
enable_testing()
add_test(NAME c11_contracts COMMAND bridge_tests)
add_test(NAME cli_demo COMMAND bridge_demo 42 8192)
add_test(NAME cli_bad_length COMMAND bridge_demo 42 16777217)
set_tests_properties(cli_bad_length PROPERTIES WILL_FAIL TRUE)
add_test(NAME cli_bad_number COMMAND bridge_demo -1 4)
set_tests_properties(cli_bad_number PROPERTIES WILL_FAIL TRUE)
